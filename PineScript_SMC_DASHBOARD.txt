add_zone(top, btm, isBull, isOb, cond, startBar) =>
    if cond and not na(top) and not na(btm) and top != btm

// 7. MERGE OVERLAPPING ZONES

zones_overlap(top1, btm1, top2, btm2) =>
    not (btm1 > top2 or btm2 > top1)

if array.size(allZones) > 1
    for i = array.size(allZones) - 1 to 1
        if i >= array.size(allZones)
            continue
        z1 = array.get(allZones, i)
        z2 = array.get(allZones, i - 1)
        if zones_overlap(z1.top, z1.btm, z2.top, z2.btm)
            size1 = math.abs(z1.top - z1.btm)
            size2 = math.abs(z2.top - z2.btm)
            if size1 >= size2
                box.delete(z2.bx)
                array.remove(allZones, i - 1)
            else
                box.delete(z1.bx)
                array.remove(allZones, i)

//────────────────────────────────────────────
// 8. ZONE ENGINE (MITIGATION, AGE, VISIBILITY)
//────────────────────────────────────────────
maxTaps = 2

if array.size(allZones) > 0
    for i = array.size(allZones) - 1 to 0
        if i >= array.size(allZones)
            continue

        z = array.get(allZones, i)
        bx   = z.bx
        top  = z.top
        btm  = z.btm
        age  = bar_index - z.startBar

        // FAILURE: require 2 closes beyond zone
        failed = false
        if age <= failWindow
            bullFail = z.isBull and close < btm and close[1] < btm
            bearFail = (not z.isBull) and close > top and close[1] > top
            if bullFail or bearFail
                failed := true

        // MITIGATION
        if not z.isMitigated
            // A zone is only "Broken" if price closes past the opposite side
            bool bullBroken = z.isBull and close < z.btm
            bool bearBroken = (not z.isBull) and close > z.top
            
            // Increment taps for touches, but only mitigate on a full breach
            if (high > z.btm and low < z.top)
                z.taps += 1
            
            if bullBroken or bearBroken or (z.taps > 5)
                z.isMitigated := true

        // DELETE OLD / FAILED
        if age > maxAge or failed
            box.delete(bx)
            array.remove(allZones, i)
            continue

        // EXTEND ACTIVE
        if not z.isMitigated
            box.set_right(bx, bar_index)

        // VISIBILITY
        visible = (z.isMitigated and showInactive) or (not z.isMitigated and showActive)
        col = z.isMitigated ? testCol : z.baseCol

        if visible
            box.set_bgcolor(bx, col)
            box.set_border_color(bx, col)
        else
            box.set_bgcolor(bx, color.new(color.gray, 100))
            box.set_border_color(bx, color.new(color.gray, 100))
